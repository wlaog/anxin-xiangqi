cmake_minimum_required(VERSION 3.22)
project(anxin_pikafish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PIKA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/pikafish/src")
file(GLOB_RECURSE PIKA_CPP CONFIGURE_DEPENDS "${PIKA_SOURCE}/*.cpp")
list(FILTER PIKA_CPP EXCLUDE REGEX "/main\\.cpp$|/universal/|/temp_builds/")
add_library(anxin_pikafish SHARED bridge.cpp ${PIKA_CPP})
target_include_directories(anxin_pikafish PRIVATE "${PIKA_SOURCE}")
target_compile_definitions(anxin_pikafish PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS ZSTD_DISABLE_ASM)
set_target_properties(anxin_pikafish PROPERTIES PREFIX "")
if(ANDROID)
  set_target_properties(anxin_pikafish PROPERTIES PREFIX "lib")
endif()
if(NOT MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_compile_definitions(anxin_pikafish PRIVATE IS_64BIT)
endif()
if(MSVC)
  target_compile_options(anxin_pikafish PRIVATE /O2 /EHsc /utf-8 /bigobj /MP4)
else()
  target_compile_options(anxin_pikafish PRIVATE -O3 -fPIC)
  if(WIN32)
    target_link_options(anxin_pikafish PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(WIN32)
  target_link_libraries(anxin_pikafish PRIVATE advapi32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(anxin_pikafish PRIVATE Threads::Threads)
endif()
if(ANDROID)
  target_link_libraries(anxin_pikafish PRIVATE log atomic)
endif()
